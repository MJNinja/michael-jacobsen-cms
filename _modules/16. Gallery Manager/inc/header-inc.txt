<?php if($moduleID == 16){ ?>
	<script type="text/javascript" language="javascript">
        /*REMOVE GALLERY*/
        function deleteGallery(bos) {
            var msg = "Are you sure you want to remove the selected Gallery? You can always recover the gallery at a later stage.";
            if (confirm(msg)) {
                eval('document.delete_gallery'+bos+'.method = "post"');
                eval('document.delete_gallery'+bos+'.action = "<?php echo $_SERVER['PHP_SELF'] ?>"');
                eval('document.delete_gallery'+bos+'.submit()');
            } else {
                return;
            }
        }
	</script>

	<script type="text/javascript" language="javascript">
        /*RECOVER SOFTWARE*/
        function recoverGallery(bos) {
            var msg = "Are you sure you want to recover the selected Gallery?";
            if (confirm(msg)) {
                eval('document.recover_gallery'+bos+'.method = "post"');
                eval('document.recover_gallery'+bos+'.action = "<?php echo $_SERVER['PHP_SELF'] ?>"');
                eval('document.recover_gallery'+bos+'.submit()');
            } else {
                return;
            }
        }
    </script>
