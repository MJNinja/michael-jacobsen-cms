    function drawRegionsMap() {

        var data = google.visualization.arrayToDataTable(<?php echo $geoChartData; ?>);

        var options = {
            colorAxis: {colors: ['#c8d8de', '#04a6df']},
            width: 1000,
            height: 500,

        };

        var chart = new google.visualization.GeoChart(document.getElementById('country_div'));

        chart.draw(data, options);
    }

        //GET DEFAULT VALUES

            formsSubmitted();

    //CLICK FUNCTIONS

    //SHOW TOTAL SUBMITTED FORMS
    function formsSubmitted(){
        //SET INPUT VALUE
        $('input[name=tabName]').val('formsSubmitted');

        //GET DEFAULT VALUES
        var form = $('select[name=forms-select]').val();

        //GET DIFFERENT CHART CONTAINER
        var lineChart = $('#lineChart-holder');
        var country_div = $('#country_div_holder');
        var piecharts = $('#piecharts-holder');

        //REMOVE CANVAS
        lineChart.empty();
        //country_div.empty();
        piecharts.empty();

        //HIDE DIV
        country_div.hide();
        piecharts.hide();

        //DISPLAY LOADER
        $('.form-loader').show();

        //DO AJAX REQUESTED
        $.ajax({
        	type: "POST",
        	url: '<?php echo $cms_root; ?>forms-manager/ajax.graphUpdates.php',
        	data: "form=" + form + "&linegraph=1",
        	success: function(datalist) {
        		//HIDE LOADER
                $(".form-loader").hide();

        		//CREATE NEW GRAPH
                $('#lineChart-holder').append('<div class="insight-information-heading">Forms Submitted</div><canvas id="lineChart"></canvas>');
        		var line = document.getElementById("lineChart").getContext("2d");
        		window.myLine = new Chart(line).Line(JSON.parse(datalist),{bezierCurve: false, scaleBeginAtZero: true});
        	}
        });
    }

    //SHOW COUNTRIES
    function countries(){
        //SET INPUT VALUE
        $('input[name=tabName]').val('countries');

        //GET DEFAULT VALUES
        var form = $('select[name=forms-select]').val();

        //GET DIFFERENT CHART CONTAINER
        var lineChart = $('#lineChart-holder');
        var country_div = $('#country_div_holder');
        var piecharts = $('#piecharts-holder');

        //REMOVE CANVAS
        lineChart.empty();
        //country_div.empty();
        piecharts.empty();

        //DISPLAY LOADER
        $('.form-loader').show();

        //HIDE LOADER
        $(".form-loader").hide();

        //HIDE DIV
        piecharts.hide();

        //SHOW COUNTRY DIV
        country_div.show();

    }

    //SHOW PIE CHARTS
    function moreDetails(){
        //SET INPUT VALUE
        $('input[name=tabName]').val('moreDetails');

        //GET DEFAULT VALUES
        var form = $('select[name=forms-select]').val();

        //GET DIFFERENT CHART CONTAINER
        var lineChart = $('#lineChart-holder');
        var country_div = $('#country_div_holder');
        var piecharts = $('#piecharts-holder');

        //REMOVE CANVAS
        lineChart.empty();
        //country_div.empty();
        piecharts.empty();

        //HIDE DIV
        country_div.hide();

        //SHOW DIV
        piecharts.show();

        //DISPLAY LOADER
        $('.form-loader').show();

        //DO AJAX REQUESTED
        $.ajax({
        	type: "POST",
        	url: '<?php echo $cms_root; ?>forms-manager/ajax.graphUpdates.php',
        	data: "form=" + form + "&piecharts=1",
        	success: function(datalist) {
                //SPLIT DATALIST INFORMATION
                dataArray = datalist.split("^");

        		//HIDE LOADER
                $(".form-loader").hide();

        		//CREATE NEW GRAPH
                $('#piecharts-holder').append('<div class="insight-information-heading-piecharts">Cities:</div><div class="insight-information-heading-piecharts">Languages:</div><div class="insight-information-heading-piecharts">Browsers:</div><div class="insight-information-heading-piecharts">Operating Systems:</div><div class="clear"></div><canvas id="city" class="piecharts"></canvas><canvas id="language" class="piecharts"></canvas><canvas id="browser" class="piecharts"></canvas><canvas id="os" class="piecharts"></canvas>');

                //CITY PIE CHART
    			var city = document.getElementById("city").getContext("2d");
    			window.myPie = new Chart(city).Doughnut(JSON.parse(dataArray[0]),{animationEasing : "easeOut",animationSteps : 50});

                //LANGUAGE PIE CHART
    			var language = document.getElementById("language").getContext("2d");
    			window.myPie = new Chart(language).Doughnut(JSON.parse(dataArray[1]),{animationEasing : "easeOut",animationSteps : 50});

                //LANGUAGE PIE CHART
    			var browser = document.getElementById("browser").getContext("2d");
    			window.myPie = new Chart(browser).Doughnut(JSON.parse(dataArray[2]),{animationEasing : "easeOut",animationSteps : 50});

                //LANGUAGE PIE CHART
    			var os = document.getElementById("os").getContext("2d");
    			window.myPie = new Chart(os).Doughnut(JSON.parse(dataArray[3]),{animationEasing : "easeOut",animationSteps : 50});
        	}
        });
    }
</script>
