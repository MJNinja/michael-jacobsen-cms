                    $('input[name=categories]').val(category_tags);
                }
            },
            //CHECKS TAGS AFTER THEY ARE REMOVED AND INSERT VALUES INTO INPUT FIELD
            afterTagRemoved: function() {

<!-- VIEW MORE BLOG POSTS JAVASCRIPT BEGIN-->
<script type="text/javascript" language="javascript">
$(function(){
   var page = 1;
   var pages = $("#pages-blog-posts").val(); //TOTAL NUMBER OF PAGES

   $("#recent-blog-posts").load("<?php echo $cms_root ?>ajax/ajax.fetchBlogPosts.php");

   //WHEN THE 'LOAD MORE' BUTTON IS PRESSED
   $("#loadmore-blog-posts").click(function(){

     //HIDE LOAD MORE BUTTON
     $('#loadmore-blog-posts').hide();

     //SHOW PRELOADER
     $("#loader-blog-posts").show();

     var next = page+=1;

     $.get("<?php echo $cms_root ?>ajax/ajax.fetchBlogPosts.php?page="+next, function(data){
       //HIDE PRELOADER
       $("#loader-blog-posts").hide();

       //SHIW LOAD MORE BUTTON
       $('#loadmore-blog-posts').show();

       if(next==pages){
         $("#loadmore-blog-posts").remove(); //IF ALL PAGES ARE LOADED, THE BUTTON WILL BE REMOVED
       }

       $("#recent-blog-posts").append(data); //LOADS THE NEW PAGE OF CONTENT UNDER THE REST

     });
   });
});
</script>
<!-- VIEW MORE BLOG POSTS JAVASCRIPT END-->
<?php }?>
