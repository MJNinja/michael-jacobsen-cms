  `eventImageFile` varchar(255) DEFAULT NULL,
  `eventImageTitle` varchar(255) DEFAULT NULL,
  `startDate` datetime DEFAULT '0000-00-00 00:00:00',
  `endDate` datetime DEFAULT '0000-00-00 00:00:00',

  `views` int(11) DEFAULT '0',
  `url` text,
  `createdBy` int(11) DEFAULT '0',
  `createdDate` datetime DEFAULT '0000-00-00 00:00:00',
  `modifiedBy` int(11) DEFAULT '0',
  `modifiedDate` datetime DEFAULT '0000-00-00 00:00:00',
