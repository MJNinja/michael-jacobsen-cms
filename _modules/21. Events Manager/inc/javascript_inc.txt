<script type="text/javascript" language="javascript">
$(function(){
   var page = 1;
   var pages = $("#pages-events").val(); //TOTAL NUMBER OF PAGES

   $("#recent-events").load("<?php echo $cms_root ?>ajax/ajax.fetchEvents.php");

   //WHEN THE 'LOAD MORE' BUTTON IS PRESSED
   $("#loadmore-events").click(function(){

     //HIDE LOAD MORE BUTTON
     $('#loadmore-events').hide();

     //SHOW PRELOADER
     $("#loader-events").show();

     var next = page+=1;

     $.get("<?php echo $cms_root ?>ajax/ajax.fetchEvents.php?page="+next, function(data){
       //HIDE PRELOADER
       $("#loader-events").hide();

       //SHIW LOAD MORE BUTTON
       $('#loadmore-events').show();

       if(next==pages){
         $("#loadmore-events").remove(); //IF ALL PAGES ARE LOADED, THE BUTTON WILL BE REMOVED
       }

       $("#recent-events").append(data); //LOADS THE NEW PAGE OF CONTENT UNDER THE REST

     });
   });
});
</script>
