<script type="text/javascript" language="javascript">
//CHECK FOR CHANEGE IN MAIN CATEGORY DROP DOWN
$('select[name=product-category]').on('change', function() {
    var ddValue = this.value;

    //CHECK IF A MAIN CATEGORY HAS BEEN SELECTED
    if(ddValue != '' && ddValue != ' ' && ddValue != 0){
        //SHOW LOADER
        $('#loader-product-cat').show();

        //DO AJAX
        $.ajax({
    		type: "POST",
    		url: "<?php echo $cms_root; ?>ajax/ajax.productSubCategories.php",
    		data: "ddValue=" + ddValue,
    		success: function(response) {
                responseArray   = response.split('#@#');
                result          = responseArray[0];
                content         = responseArray[1];

                //IF SUCCESSDFUL
                if(result == 'success'){
                    //APPEND DEFAULT
                    $('select[name=product-sub-category]').empty().append('<option value="0">-- Select a Sub Category --</option>');
                    //APPEND CONTENT
                    $('select[name=product-sub-category]').append(content);
                    //SHOW SUB CATEGORY DROP DOWN
                    $('.product-sub-cat-holder').show();
                }
                //IF FAILED
                else{
                    //HIDE SUB CATEGORY DROP DOWN
                    $('.product-sub-cat-holder').hide();
                }

                //HIDE LOADER
                $('#loader-product-cat').hide();
            }
    	});
    }else{
        //HIDE SUB CATEGORY DROP DOWN
        $('.product-sub-cat-holder').hide();

        //HIDE LOADER
        $('#loader-product-cat').hide();
    }
});
</script>

$(function(){
   var page = 1;
   var pages = $("#pages-products").val(); //TOTAL NUMBER OF PAGES

   //WHEN THE 'LOAD MORE' BUTTON IS PRESSED
   $("#loadmore-products").click(function(){

     //HIDE LOAD MORE BUTTON

     //SHOW PRELOADER
     $("#loader-products").show();

     var next = page+=1;

     $.get("<?php echo $cms_root ?>ajax/ajax.fetchProducts.php?page="+next, function(data){
       //HIDE PRELOADER
       $("#loader-products").hide();

       //SHIW LOAD MORE BUTTON
       $('#loadmore-products').show();

       if(next==pages){
         $("#loadmore-products").remove(); //IF ALL PAGES ARE LOADED, THE BUTTON WILL BE REMOVED
       }

       $("#recent-products").append(data); //LOADS THE NEW PAGE OF CONTENT UNDER THE REST

     });
   });
});
</script>
