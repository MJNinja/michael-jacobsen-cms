CREATE TABLE IF NOT EXISTS `staff_members` (
  `staffID` int(11) NOT NULL AUTO_INCREMENT,
  `staffName` varchar(255) DEFAULT NULL,
  `staffSurname` varchar(255) DEFAULT NULL,
  `staffPosition` varchar(255) DEFAULT NULL,
  `staffImageTitle` varchar(255) DEFAULT NULL,
  `staffImage` varchar(255) DEFAULT NULL,
  `staffEmail` varchar(255) DEFAULT NULL,
  `staffContact` varchar(255) DEFAULT NULL,
  `staffDescription` text,
  `createdBy` int(11) DEFAULT '0',
  `createdDate` datetime DEFAULT '0000-00-00 00:00:00',
  `modifiedBy` int(11) DEFAULT '0',
  `modifiedNumber` int(11) DEFAULT '0',
  `modifiedDate` datetime DEFAULT '0000-00-00 00:00:00',
  `deletedBy` int(11) DEFAULT '0',
  `deletedDate` datetime DEFAULT '0000-00-00 00:00:00',
  `sequence` int(11) DEFAULT '0',
  PRIMARY KEY (`staffID`)
);
