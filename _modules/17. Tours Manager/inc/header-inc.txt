        function deleteTour(bos) {
            var msg = "Are you sure you want to remove the selected Tour? Once it has been deleted it can't be recovered.";
            if (confirm(msg)) {
                eval('document.delete_tour'+bos+'.method = "post"');
                eval('document.delete_tour'+bos+'.action = "<?php echo $_SERVER['PHP_SELF'] ?>"');
                eval('document.delete_tour'+bos+'.submit()');
            } else {
                return;
            }
        }
	</script>
