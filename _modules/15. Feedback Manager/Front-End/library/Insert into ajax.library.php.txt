	//GET FEEDBACK COOKIE VALUE
	//#################################################
	function getFeedbackCookieValue($fbcn, $fbpusc){
		//CONNECT TO ENCRYPT/DECRYPT CLASS
        $encryptDecrypt = new encryptDecrypt();

        //DEFINE ENCRYPTION_KEY
    	define("ENCRYPTION_KEY", "%@#!^&$*");

        //DECRYPT COOKIE
        $fbcd   = $encryptDecrypt->decrypt($_COOKIE[$fbcn], ENCRYPTION_KEY);
        $fbpusc = $encryptDecrypt->decrypt($_COOKIE[$fbpusc], ENCRYPTION_KEY);

        //EXPLODE RESULT
        $cookieDataArray    = explode("#@#", $fbcd);
        $cookieDataShow     = $cookieDataArray[0];

        //RETURN FEEDBACK DATA
        if($cookieDataShow == 'yes' || $cookieDataShow == 'no'){
            return $fbcd.'#@#'.$fbpusc;
        }else{
            return ' ';
        }
	}

	//CREATE FEEDBACK COOKIE
	//#################################################
	function createFeedbackCookie($fbcn){
		//CONNECT TO ENCRYPT/DECRYPT CLASS
        $encryptDecrypt = new encryptDecrypt();

        //DEFINE ENCRYPTION_KEY
    	define("ENCRYPTION_KEY", "%@#!^&$*");

        //DEFAULT VARIABLE
        $currentDateMilliseconds    = round(microtime(true) * 1000);

        //DEFAULT VARIABLES
        $cexpire    = 1;
        $cname      = $fbcn;
        $cvalue     = "yes#@#".$currentDateMilliseconds;

        //ENCRYPT COOKIE
        $cvalue = $encryptDecrypt->encrypt($cvalue, ENCRYPTION_KEY);

        //CREATE COOKIE
        setcookie($cname, $cvalue, time() + (86400 * $cexpire), "/"); // 86400 = 1 day
	}

	//GET FEEDBACK COOKIE VALUE
	//#################################################
	function updateFeedbackCookie($fbcn){
		//CONNECT TO ENCRYPT/DECRYPT CLASS
        $encryptDecrypt = new encryptDecrypt();

        //DEFINE ENCRYPTION_KEY
    	define("ENCRYPTION_KEY", "%@#!^&$*");

        //DECRYPT COOKIE
        $fbcd = $encryptDecrypt->decrypt($_COOKIE[$fbcn], ENCRYPTION_KEY);

        //EXPLODE RESULT
        $cookieDataArray    = explode("#@#", $fbcd);
        $cookieDataTime     = $cookieDataArray[1];

        //DEFAULT VARIABLES
        $cexpire    = 60;
        $cname      = $fbcn;
        $cvalue     = "no#@#".$cookieDataTime;

        //ENCRYPT COOKIE
        $cvalue = $encryptDecrypt->encrypt($cvalue, ENCRYPTION_KEY);

        //DESTROY COOKIE
        setcookie($cname, '', 1, "/");
        unset($_COOKIE[$cname]);

        //UPDATE COOKIE
        setcookie($cname, $cvalue, time() + (86400 * $cexpire), "/");
	}
