<script type="text/javascript" language="javascript">
$(function(){
   var page = 1;
   var pages = $("#pages-feedback-messages").val(); //TOTAL NUMBER OF PAGES

   $("#recent-feedback-messages").load("<?php echo $cms_root ?>ajax/ajax.fetchFeedbackMessages.php");

   //WHEN THE 'LOAD MORE' BUTTON IS PRESSED
   $("#loadmore-feedback-messages").click(function(){

     //HIDE LOAD MORE BUTTON
     $('#loadmore-feedback-messages').hide();

     //SHOW PRELOADER
     $("#loader-feedback-messages").show();

     var next = page+=1;

     $.get("<?php echo $cms_root ?>ajax/ajax.fetchFeedbackMessages.php?page="+next+"&f=<?php echo $field; ?>", function(data){
       //HIDE PRELOADER
       $("#loader-feedback-messages").hide();

       //SHIW LOAD MORE BUTTON
       $('#loadmore-feedback-messages').show();

       if(next==pages){
         $("#loadmore-feedback-messages").remove(); //IF ALL PAGES ARE LOADED, THE BUTTON WILL BE REMOVED
       }

       $("#recent-feedback-messages").append(data); //LOADS THE NEW PAGE OF CONTENT UNDER THE REST

     });
   });
});
</script>
